	cookingtime = {30 mins},
	difficulty = {Beginner},
	origin = {USA},
	tags = {Breakfast, Quick},
	ingredients = {
			\ingredient{bacon}[4][slices]
			\ingredient{potatoes}[2][large][peeled and chopped (2½ \cu)]
